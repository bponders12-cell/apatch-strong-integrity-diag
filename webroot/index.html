GitHub Actions Release Workflow (.github/workflows/release.yml)

Validates POSIX sh -n syntax on every push and automatically attaches apatch_strong_integrity_diag_v*.zip to GitHub Releases when you push a v* tag.

name: Build & Release APatch Module

on:
  push:
    branches: [ main, master ]
    tags: [ 'v*' ]
  pull_request:
    branches: [ main, master ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build-and-validate:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Validate Shell Scripts (POSIX sh -n)
        run: |
          sh -n customize.sh
          sh -n action.sh
          sh -n scripts/diag_engine.sh
          sh -n scripts/apply_fix.sh

      - name: Read Version from module.prop
        id: meta
        run: |
          VER=$(grep '^version=' module.prop | cut -d= -f2 | tr -d '\r\n')
          ZIP_NAME="apatch_strong_integrity_diag_${VER}.zip"
          echo "version=${VER}" >> "$GITHUB_OUTPUT"
          echo "zip_name=${ZIP_NAME}" >> "$GITHUB_OUTPUT"

      - name: Package Flashable APatch Module ZIP
        run: |
          chmod 0755 action.sh customize.sh scripts/diag_engine.sh scripts/apply_fix.sh
          zip -r9 "${{ steps.meta.outputs.zip_name }}" \
            META-INF scripts webroot module.prop customize.sh action.sh README.md LICENSE

      - name: Upload Module Build Artifact
        uses: actions/upload-artifact@v4
        with:
          name: ${{ steps.meta.outputs.zip_name }}
          path: ${{ steps.meta.outputs.zip_name }}

      - name: Publish GitHub Release (on Tag Push)
        if: startsWith(github.ref, 'refs/tags/v')
        uses: softprops/action-gh-release@v2
        with:
          files: |
            ${{ steps.meta.outputs.zip_name }}
            CHANGELOG.md
          generate_release_notes: true
